cmake_minimum_required(VERSION 3.16)
project(mini_dbms VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(DB_SOURCES src/abi.c src/error.c src/platform.c src/bptree.c src/parser.c
    src/storage.c src/engine.c src/script.c)
add_library(minidb STATIC ${DB_SOURCES})
add_library(minidb_shared SHARED ${DB_SOURCES})
set_target_properties(minidb_shared PROPERTIES OUTPUT_NAME minidb ARCHIVE_OUTPUT_NAME minidb_import)
target_compile_definitions(minidb_shared PRIVATE MINIDB_SHARED)
if(NOT MSVC)
    find_package(Threads REQUIRED)
endif()
foreach(db_target minidb minidb_shared)
    target_include_directories(${db_target} PUBLIC include PRIVATE src)
    if(MSVC)
        target_compile_definitions(${db_target} PUBLIC _CRT_SECURE_NO_WARNINGS)
        target_compile_options(${db_target} PRIVATE /W4 /WX)
    else()
        target_link_libraries(${db_target} PUBLIC Threads::Threads)
        target_compile_definitions(${db_target} PUBLIC _POSIX_C_SOURCE=200809L _FILE_OFFSET_BITS=64)
        target_compile_options(${db_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()

add_executable(mini_db src/main.c)
target_link_libraries(mini_db PRIVATE minidb)
add_executable(db_tests tests/test_main.c)
target_include_directories(db_tests PRIVATE src)
target_link_libraries(db_tests PRIVATE minidb)
add_executable(db_bench tools/benchmark.c)
target_include_directories(db_bench PRIVATE src)
target_link_libraries(db_bench PRIVATE minidb)
add_executable(db_load tools/load_benchmark.c)
target_include_directories(db_load PRIVATE src)
target_link_libraries(db_load PRIVATE minidb)
enable_testing()
add_test(NAME engine_tests COMMAND db_tests)
set_tests_properties(engine_tests PROPERTIES TIMEOUT 60)
